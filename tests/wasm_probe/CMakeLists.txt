cmake_minimum_required(VERSION 3.22)
project(RetroPaintWasmProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "Use an Emscripten Qt toolchain for the Wasm probe")
endif()
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Gui Widgets)
get_target_property(qtFeatures Qt6::Core QT_ENABLED_PUBLIC_FEATURES)
if(NOT "wasm_exceptions" IN_LIST qtFeatures)
  message(FATAL_ERROR "The Qt SDK must be built with -feature-wasm-exceptions")
endif()
qt_add_executable(wasm_probe main.cpp ../../src/document.cpp)
target_include_directories(wasm_probe PRIVATE ../../src)
target_link_libraries(wasm_probe PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
target_compile_options(wasm_probe PRIVATE -fwasm-exceptions -Wall -Wextra -Wpedantic)
target_link_options(wasm_probe PRIVATE -fwasm-exceptions -sALLOW_MEMORY_GROWTH=1 -sEXIT_RUNTIME=0)
set_target_properties(wasm_probe PROPERTIES QT_WASM_INITIAL_MEMORY "64MB")
set(fontFile ../../assets/fonts/NanumGothic-Regular.ttf)
set_source_files_properties(${fontFile} PROPERTIES QT_RESOURCE_ALIAS NanumGothic.ttf)
qt_add_resources(wasm_probe probe_font PREFIX "/" FILES ${fontFile})
qt_add_executable(wasm_performance performance.cpp ../../src/document.cpp)
target_include_directories(wasm_performance PRIVATE ../../src)
target_link_libraries(wasm_performance PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
target_compile_options(wasm_performance PRIVATE -fwasm-exceptions -Wall -Wextra -Wpedantic)
target_link_options(wasm_performance PRIVATE -fwasm-exceptions -sALLOW_MEMORY_GROWTH=1 -sMAXIMUM_MEMORY=536870912 -sEXIT_RUNTIME=0)
set_target_properties(wasm_performance PROPERTIES QT_WASM_INITIAL_MEMORY "64MB")
