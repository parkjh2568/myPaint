cmake_minimum_required(VERSION 3.22)
project(RetroPaint VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets)
add_library(paintcore src/document.cpp)
target_include_directories(paintcore PUBLIC src)
target_link_libraries(paintcore PUBLIC Qt6::Core Qt6::Gui)
add_library(paintui src/canvas.cpp src/mainwindow.cpp)
target_link_libraries(paintui PUBLIC paintcore Qt6::Widgets)
if(EMSCRIPTEN)
  get_target_property(qtFeatures Qt6::Core QT_ENABLED_PUBLIC_FEATURES)
  if(NOT "wasm_exceptions" IN_LIST qtFeatures)
    message(FATAL_ERROR "Use a Qt Wasm SDK built with -feature-wasm-exceptions")
  endif()
  target_sources(paintui PRIVATE src/mainwindow_web.cpp src/browserbridge.cpp)
  target_compile_options(paintcore PUBLIC -fwasm-exceptions)
  target_link_options(paintcore PUBLIC -fwasm-exceptions)
  qt_add_executable(retropaint src/webmain.cpp)
  set(fontFile assets/fonts/NanumGothic-Regular.ttf)
  set_source_files_properties(${fontFile} PROPERTIES QT_RESOURCE_ALIAS NanumGothic.ttf)
  qt_add_resources(retropaint web_font PREFIX "/" FILES ${fontFile})
  target_link_options(retropaint PRIVATE "--pre-js=${CMAKE_CURRENT_SOURCE_DIR}/web/browser.js"
    -sALLOW_MEMORY_GROWTH=1 -sMAXIMUM_MEMORY=536870912 -sABORTING_MALLOC=0 -sEXIT_RUNTIME=0
    "-sEXPORTED_FUNCTIONS=['_main','_malloc','_free','_retropaint_browser_result','_retropaint_browser_drop']")
  set_target_properties(retropaint PROPERTIES QT_WASM_INITIAL_MEMORY "64MB")
  set_property(TARGET retropaint APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/web/browser.js")
else()
  add_executable(retropaint WIN32 MACOSX_BUNDLE src/main.cpp)
endif()
target_link_libraries(retropaint PRIVATE paintui)
set_target_properties(retropaint PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER org.retropaint.desktop MACOSX_BUNDLE_BUNDLE_NAME RetroPaint)
if(MSVC)
  target_compile_options(paintcore PRIVATE /W4)
  target_compile_options(paintui PRIVATE /W4 /utf-8)
  target_compile_options(paintcore PRIVATE /utf-8)
  target_compile_options(retropaint PRIVATE /W4)
  target_compile_options(retropaint PRIVATE /utf-8)
else()
  target_compile_options(paintcore PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(paintui PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(retropaint PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING AND NOT EMSCRIPTEN)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME web_stage COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_stage_web.py)
  endif()
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE paintcore Qt6::Test)
  add_test(NAME core COMMAND core_tests)
  add_executable(ui_tests tests/ui_tests.cpp)
  target_link_libraries(ui_tests PRIVATE paintui Qt6::Test)
  add_executable(perf_probe tests/perf_probe.cpp)
  target_link_libraries(perf_probe PRIVATE paintcore)
  add_executable(allocation_tests tests/allocation_tests.cpp)
  target_link_libraries(allocation_tests PRIVATE paintcore)
  add_test(NAME allocation_failure COMMAND allocation_tests)
  add_test(NAME smoke COMMAND retropaint --smoke)
  if(MSVC)
    target_compile_options(core_tests PRIVATE /utf-8)
    target_compile_options(ui_tests PRIVATE /utf-8)
  endif()
  add_test(NAME ui COMMAND ui_tests)
  if(MSVC)
    target_compile_options(core_tests PRIVATE /W4)
    target_compile_options(ui_tests PRIVATE /W4)
    target_compile_options(perf_probe PRIVATE /W4 /utf-8)
    target_compile_options(allocation_tests PRIVATE /W4 /utf-8)
  else()
    target_compile_options(core_tests PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(ui_tests PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(perf_probe PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(allocation_tests PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  set_tests_properties(core ui PROPERTIES ENVIRONMENT QT_QPA_PLATFORM=offscreen TIMEOUT 90)
  set_tests_properties(smoke PROPERTIES ENVIRONMENT QT_QPA_PLATFORM=offscreen TIMEOUT 20
    RESOURCE_LOCK retropaint_smoke_session)
  set_tests_properties(allocation_failure PROPERTIES TIMEOUT 10)
  set_tests_properties(ui PROPERTIES RESOURCE_LOCK retropaint_test_recovery)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_library(network_guard SHARED tests/network_guard.cpp)
    target_link_libraries(network_guard PRIVATE ${CMAKE_DL_LIBS})
    add_test(NAME offline_smoke COMMAND ${CMAKE_COMMAND} -E env
      "LD_PRELOAD=$<TARGET_FILE:network_guard>" "QT_QPA_PLATFORM=offscreen"
      $<TARGET_FILE:retropaint> --smoke)
    set_tests_properties(offline_smoke PROPERTIES TIMEOUT 15 RESOURCE_LOCK retropaint_smoke_session)
    add_executable(network_probe tests/network_probe.cpp)
    add_test(NAME network_guard_selftest COMMAND ${CMAKE_COMMAND}
      "-DGUARD=$<TARGET_FILE:network_guard>" "-DPROBE=$<TARGET_FILE:network_probe>"
      -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_network_guard.cmake)
    set_tests_properties(network_guard_selftest PROPERTIES TIMEOUT 5)
    add_test(NAME offline_ui COMMAND ${CMAKE_COMMAND} -E env
      "LD_PRELOAD=$<TARGET_FILE:network_guard>" "QT_QPA_PLATFORM=offscreen"
      $<TARGET_FILE:ui_tests>)
    set_tests_properties(offline_ui PROPERTIES TIMEOUT 90)
    set_tests_properties(offline_ui PROPERTIES RESOURCE_LOCK retropaint_test_recovery)
  endif()
endif()
install(TARGETS retropaint BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES LICENSE NOTICE.md DESTINATION ${CMAKE_INSTALL_DATADIR}/retropaint)
option(RETROPAINT_DEPLOY_RUNTIME "Collect Qt runtime dependencies for a native deployment" OFF)
if(RETROPAINT_DEPLOY_RUNTIME AND NOT EMSCRIPTEN)
  if(Qt6_VERSION VERSION_LESS 6.5)
    qt_generate_deploy_app_script(TARGET retropaint FILENAME_VARIABLE deployScript)
  else()
    qt_generate_deploy_app_script(TARGET retropaint OUTPUT_SCRIPT deployScript)
  endif()
  install(SCRIPT ${deployScript})
endif()
